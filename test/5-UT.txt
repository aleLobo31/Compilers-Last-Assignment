int a = 0, b = 2, c;

main() {
    // CEV6.1: Bucle While con Sentencia Simple
    puts("--- Prueba 6.1 ---");
    while (a < 3) {
        a = a + 1;
    }

    // CEV6.2: Bucle While Anidado
    // Obligamos al parser a suspender el bucle exterior, 
    // resolver el interior, y luego retomar el exterior.
    puts("--- Prueba 6.2 ---");
    while (b > 0) {
        c = 0;
        
        while (c < 2) {
            printf("Anidado: ", c);
            c = c + 1;
        }
        
        b = b - 1;
    }

    // CEV6.3: Bucle While Vacío
    // Comprueba el caso base de tu regla `lista_sentencias` (cadena vacía)
    puts("--- Prueba 6.3 ---");
    while (a == 100) {
    }
}
