int a = 10, b = 5;

main() {
    int r;
    
    // CEV5.1 (&&) y CEV5.2 (||)
    r = (a > 0) && (b < 10);
    r = (a == 0) || (b == 5);
    
    // CEV5.3 (!=) y CEV5.4 (==)
    r = a != b;
    r = a == 10;
    
    // CEV5.5 (<) y CEV5.6 (>)
    r = a < b;
    r = a > b;
    
    // CEV5.7 (<=) y CEV5.8 (>=)
    r = a <= 10;
    r = b >= 5;
    
    // CEV5.9 (%)
    r = a % b;
}
