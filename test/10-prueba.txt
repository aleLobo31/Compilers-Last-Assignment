int limite = 10;

main() {
    int a = 5;
    
    // Prueba IF con ELSE y multiples sentencias
    if (a < limite) {
        puts("Es menor");
        a = a + 1;
    } else {
        puts("Es mayor o igual");
        a = 0;
    }
    
    // Prueba IF sin ELSE
    if (a == 6) {
        puts("Incremento completado correctamente");
    }
}
